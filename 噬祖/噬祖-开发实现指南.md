## 四、难度系统（核心）

### 4.1 战力计算

```typescript
/** 策划折算：三属性归一化取平均 × 基因锁加成 × 装备加成。基础值 → 战力 1 */
function computePower(p: SaveData['player']): number {
  const atkTerm  = 1 + p.permAtkPct  / 100;          // 基础攻击 10
  const hpTerm   = 1 + p.permHpPct   / 100;          // 基础生命 100
  const spdTerm  = 1 + p.permSpeedPct / 100;         // 基础速度 220
  const geneBonus = geneLockPowerBonus(p.geneLocks); // 每段基因锁少量加成（见平衡表）
  const gearBonus = gearPowerBonus(p.gear);          // 装备加成（见 11 章）
  return ((atkTerm + hpTerm + spdTerm) / 3) * geneBonus * gearBonus;
}
```

### 4.2 D 值计算

```typescript
const DIFFICULTY_COEF: Record<DifficultyLevel, number> = {
  easy: 0.9, normal: 1.5, hard: 2.0,
};

/** 副本难度值 D = 玩家当前战力 × 难度等级系数 */
function dungeonDifficulty(power: number, level: DifficultyLevel): number {
  return power * DIFFICULTY_COEF[level];
}
```

### 4.3 难度系数调整时机（每次副本结算后调用）

```typescript
interface RunResult { victory: boolean; }

/**
 * 规则（整体策划 3.2 / 5.1）：
 * 通关   → 下次 dynFactor ×(1.05~1.15) 累乘
 * 连败2-3→ dynFactor ×(0.90~0.95)
 * 钳制   → [0.70, 1.50]
 */
function adjustDynamicFactor(save: SaveData, result: RunResult): { deltaPct: number } {
  const p = save.player;
  if (result.victory) {
    const mult = randRange(1.05, 1.15);          // 随机取 +5%~+15%
    p.dynFactor = clamp(p.dynFactor * mult, 0.70, 1.50);
    p.consecFails = 0;
    return { deltaPct: Math.round((mult - 1) * 100) };
  }
  p.consecFails += 1;
  if (p.consecFails >= 2) {                       // 连败 2-3 次触发
    const mult = randRange(0.90, 0.95);
    p.dynFactor = clamp(p.dynFactor * mult, 0.70, 1.50);
    p.consecFails = 0;
  }
  return { deltaPct: -Math.round((1 - p.dynFactor) * 100) };  // UI 提示用
}
```

### 4.4 敌人数值生成（运行时）

```typescript
/** 阶段系数（整体策划 3.2 / 关卡策划三章） */
const STAGE_COEF = [0.9, 1.0, 1.15, 1.3];          // 阶段 1-4
const BOSS_COEF_MIN = 1.10, BOSS_COEF_MAX = 1.15;  // 阶段 5 位面之主

interface UnitTemplate { baseHp: number; baseAtk: number; }  // 战力 1 基准

interface EnemyStats { hp: number; atk: number; }

/** 敌人实际数值 = 基准 × D × 阶段系数 × 动态难度系数 */
function buildEnemy(unit: UnitTemplate, save: SaveData, stage: 1|2|3|4|5): EnemyStats {
  const D = dungeonDifficulty(computePower(save.player), save.player.difficultyLevel);
  let coef: number;
  if (stage === 5) {
    coef = randRange(BOSS_COEF_MIN, BOSS_COEF_MAX);  // 锁定"差一点"
  } else {
    coef = STAGE_COEF[stage - 1];
  }
  const d = D * coef * save.player.dynFactor;
  return {
    hp:  Math.ceil(unit.baseHp * d),
    atk: Math.ceil(unit.baseAtk * d),
  };
}

/** 位面类型差异（关卡策划三章补充说明） */
function applySpawnStyleAdjust(plane: PlaneConfig, base: UnitTemplate, waveCount: number) {
  if (plane.spawnStyle === 'horde')   return { unit: { ...base, baseHp: base.baseHp * 0.75 }, waveCount: waveCount + 1 };
  if (plane.spawnStyle === 'single')  return { unit: { ...base, baseHp: base.baseHp * 1.5  }, waveCount: waveCount - 1 };
  return { unit: base, waveCount };
}
```

### 4.5 伤害公式

```typescript
/** 单次伤害 = 攻击力 × 技能倍率 × 暴击(1.5×) − 目标防御(若有) */
function calcDamage(atk: number, skillMul: number, isCrit: boolean, def: number = 0): number {
  return Math.max(1, atk * skillMul * (isCrit ? 1.5 : 1) - def);
}
```

## 五、位面随机池（权重抽取）

### 5.1 权重规则

```typescript
/**
 * 权重（整体策划 3.4 / 关卡策划五·规则1）：
 *   已激活路线位面 ×2（刷技能导向）
 *   未激活路线位面 ×1（首进激活价值）
 *   互斥路线位面   ×0（不可抽中）
 */
function planeWeight(plane: PlaneConfig, p: SaveData['player']): number {
  // 诸天之心：不激活路线 → 权重始终 = 1（可被已激活路线加权稀释，属正常）
  if (plane.id === 'zhutian') return 1;
  let w = 0;
  for (const r of plane.routes) {
    if (p.sealedRoutes.includes(r)) return 0;            // 互斥 → 0
    if ((p.geneLocks[r] ?? 0) >= 1) { w += 2; continue; }// 已激活 → 贡献 2
    w += 1;                                              // 未激活 → 贡献 1
  }
  return w;
}
```

### 5.2 加权随机抽取（含首进固定与互斥兜底）

```typescript
/** 开裂缝：返回本次副本位面 id */
function rollPlane(save: SaveData, rng: RNG): PlaneId {
  // 规则 3：首次副本固定机关城（教学）
  if (save.player.totalRuns === 0) return 'jiguan';

  const pool = Object.values(PLANES).filter(p => planeWeight(p, save.player) > 0);
  // 规则 2 兜底：若过滤后为空（理论不会发生，防御性处理）→ 全池重随机
  const safePool = pool.length > 0 ? pool : Object.values(PLANES);

  // 加权随机（权重累计法）
  const total = safePool.reduce((s, p) => s + planeWeight(p, save.player), 0);
  let roll = rng.next() * total;
  for (const p of safePool) {
    roll -= planeWeight(p, save.player);
    if (roll <= 0) return p.id;
  }
  return safePool[safePool.length - 1].id;
}

/** 互斥冲突兜底（规则 2）：已激活渡劫时，尸海末世权重已为 0，天然不抽中；
 *  若极端情况仍命中（配置错误）→ 替换为同组相容位面 */
function resolveConflict(planeId: PlaneId, p: SaveData['player'], rng: RNG): PlaneId {
  const plane = PLANES[planeId];
  const hasConflict = plane.routes.some(r => p.sealedRoutes.includes(r));
  if (!hasConflict) return planeId;
  const compatible = Object.values(PLANES)
    .filter(p2 => p2.id !== planeId && planeWeight(p2, p) > 0);
  if (compatible.length > 0) {
    return compatible[Math.floor(rng.next() * compatible.length)].id;
  }
  // 防御兜底（正常不会走到）：从全池随机一个
  const all = Object.values(PLANES);
  return all[Math.floor(rng.next() * all.length)].id;
}
```

```typescript
/** 三选一生成：匹配位面→路线技能池；不匹配→通用属性池 */
function rollUpgradeOptions(plane: PlaneConfig, save: SaveData, rng: RNG): UpgradeOption[] {
  const p = save.player;
  const isMatched = plane.routes.some(r => (p.geneLocks[r] ?? 0) >= 1);

  let pool: UpgradeOption[];
  if (isMatched) {
    // 该路线基因锁分支技能池（已解锁段位的技能 + 本局可学）
    const route = plane.routes.find(r => (p.geneLocks[r] ?? 0) >= 1)!;
    pool = routeSkillPool(route, p);                 // 见 skills.ts
  } else {
    pool = GENERIC_ATTR_POOL;                        // 通用属性池（见数值平衡表六章）
  }

  // 稀有度权重：基础40% / 特色40% / 稀有15% / 传说5%（仅匹配位面产出传说）
  return weightedPickNoRepeat(pool, 3, rng);
}

/** 三通道边界：装备不走三选一，走击杀掉落（rollCommonGear / rollBossGear，见 9/11 章）。
 *  不匹配位面零技能，但装备掉率 ×1.5——装备是全位面共享的第三通道。 */

/** 普适加权随机（去重） */
function weightedPickNoRepeat<T extends { rarity: string }>(
  pool: T[], count: number, rng: RNG
): T[] {
  const RARITY_WEIGHT = { base: 40, feature: 40, rare: 15, legend: 5 };
  const out: T[] = [];
  const remain = [...pool];
  while (out.length < count && remain.length > 0) {
    const wOf = (x: T) => RARITY_WEIGHT[x.rarity] ?? 1;
    const total = remain.reduce((s, x) => s + wOf(x), 0);
    let roll = rng.next() * total;
    let idx = 0;
    for (let i = 0; i < remain.length; i++) {
      roll -= wOf(remain[i]);
      if (roll <= 0) { idx = i; break; }
    }
    out.push(remain.splice(idx, 1)[0]);
  }
  return out;
}
```

## 九、掉落与 PRD 保底

```typescript
/** 伪随机计数器（整体策划 4.5：连续不触发则概率递增） */
class PRDCounter {
  private count = 0;
  constructor(private baseP: number, private step: number) {}
  roll(rng: RNG): boolean {
    const p = Math.min(this.baseP + this.step * this.count, 1);
    if (rng.next() < p) { this.count = 0; return true; }
    this.count += 1;
    return false;
  }
}

/** 位面之主掉落判定 */
function rollBossDrop(plane: PlaneConfig, save: SaveData, rng: RNG): DropResult {
  const drops: string[] = ['genes'];               // 必掉基因
  // 传承：普通 20%（保底 8 次）/ 稀有 5%（保底 30-40 次，1/概率×1.5-2）
  const relicP = plane.id === 'zhutian' ? 1 : 0.20; // 诸天之心必掉「崩坏之心」
  if (relicP === 1 || new PRDCounter(relicP, relicP * 2).roll(rng)) {
    drops.push(plane.id === 'zhutian' ? 'relic_benghuaixin' : `relic_${plane.id}`);
  }
  // 超低概率技能（传说）：0.5%，PRD 递增，仅匹配位面
  if (planeChannel(save, plane) === 'skill' && new PRDCounter(0.005, 0.001).roll(rng)) {
    drops.push('legend_skill');
  }
  // 隐藏技能（禁忌级）：0.1%，PRD 递增，仅匹配位面；槽位永久替换
  if (planeChannel(save, plane) === 'skill' && new PRDCounter(0.001, 0.0004).roll(rng)) {
    drops.push('hidden_skill');                    // 具体技能见 rollHiddenSkill（10 章）
  }
  // 装备：位面之主 100% 保底稀有蓝（紫 25% / 金 5%）；不匹配位面掉率 ×1.5
  const gearRarity = rollBossGear(plane, save, rng);
  if (gearRarity) drops.push(`gear:${gearRarity}`);
  return { drops, genes: randRange(200, 300) };
}

/** 位面之主装备稀有度：保底蓝（紫 25% / 金 5%）；属性通道 ×1.5（见数值平衡表 7.5） */
function rollBossGear(plane: PlaneConfig, save: SaveData, rng: RNG): GearRarity | null {
  const isAttrChannel = planeChannel(save, plane) === 'attr';
  const mult = isAttrChannel ? 1.5 : 1;
  const roll = rng.next() * mult;
  if (roll < 0.05) return 'gold';                  // 金 5%
  if (roll < 0.30) return 'purple';                // 紫 25%
  return 'blue';                                   // 保底蓝
}

/** 非位面之主掉落：小怪 0.5% 白 / 精英 4% 白 1% 绿 / 阶段BOSS 8% 白 3% 绿 0.5% 蓝（×1.5 属性通道） */
function rollCommonGear(kind: 'minion' | 'elite' | 'stageBoss', isAttrChannel: boolean, rng: RNG): GearItem | null {
  const mult = isAttrChannel ? 1.5 : 1;
  const base: Record<string, [number, GearRarity]> = {
    minion:     [0.005, 'white'],
    elite:      [0.04, 'white'],
    stageBoss:  [0.08, 'white'],
  };
  const [p, r] = base[kind];
  if (rng.next() < p * mult) return generateGear(rng, r);  // 词条 roll 见 11 章
  return null;
}
```

```typescript
                      skill: { id: string; kind: 'active' | 'passive' },

  const choice = await showReplaceDialog(slotKey, skill, slot);

  return 'rejected';
}

  const slots = save.player.skillSlots;
  if (kind === 'active') {
    if (!slots.activeA) return 'activeA';
    if (!slots.activeB) return 'activeB';
    return 'activeA';                                  // 都满 → 交给 onLearnSkill 弹窗
  }
  if (!slots.passiveC) return 'passiveC';
  if (!slots.passiveD) return 'passiveD';

}
```

```typescript
function generateGear(rng: RNG, rarity: GearRarity, slot?: GearSlotId): GearItem {
  const s = slot ?? randomSlot(rng);
  const cfg = GEAR_RARITY[rarity];
  const pool = [...GEAR_SLOTS[s].affixPool];
  const affixes: GearAffix[] = [];
  while (affixes.length < cfg.affixCount && pool.length > 0) {
    const i = Math.floor(rng.next() * pool.length);
    const key = pool.splice(i, 1)[0];
    const base = AFFIX_BASE[key];
    affixes.push({ key, value: round1(base.value * cfg.mult), weight: base.weight });
  }
  return { uid: genUid(), slot: s, rarity, star: 0, affixes, name: `${cfg.name}·${GEAR_SLOTS[s].name}` };
}

function gearPowerBonus(gear: SaveData['player']['gear']): number {
  const worn = Object.values(gear).filter(Boolean) as GearItem[];
  let bonus = 1;
  for (const g of worn) {
    const mult = GEAR_RARITY[g.rarity].mult * (1 + 0.1 * g.star);  // 强化 +1 星 ×1.1
    bonus += g.affixes.length * mult * 0.025;

  return bonus;
}
```

```typescript
function onStageCleared(dungeon: DungeonBlueprint, stageNo: number) {
  if (stageNo === 5) { onDungeonCleared(dungeon); return; }
  maybeRollChoice(dungeon);          // 每阶段 BOSS 击破 + 吞噬阈值 → 三选一
}

function onPlayerDeath(dungeon: DungeonBlueprint, save: SaveData) {
  // 广告复活 1 次：复活后 1.5s 无敌 + 半血
  if (!usedAdRevive) { offerAdRevive(); return; }
  finalizeRun(dungeon, { victory: false }, save);
}
```

### 13.3 结算流程

```typescript
function finalizeRun(dungeon: DungeonBlueprint, result: RunResult, save: SaveData) {
  const p = save.player;
  p.totalRuns += 1;
  if (result.victory) p.wins += 1;

  // 1) 首进激活：本次位面路线未激活 → 永久激活（互斥封印随之生效）
  for (const r of dungeon.plane.routes) {
    if (r && (p.geneLocks[r] ?? 0) === 0 && !p.sealedRoutes.includes(r)) {
      activateRoute(save, r);
    }
  }
  // 2) 永久属性转化：局内吞噬基因 ×5% → 攻+2%/血+2%/速+1%（单局上限合计+3%）
  applyPermGrowth(save, runGenes);
  // 3) 传承 / 传说技能入库（PRD）
  // 4) 装备入库：位面之主保底 + 精英/阶段BOSS 概率（不匹配位面 ×1.5，见 9 章 rollCommonGear）
  const drops = collectGearDrops(dungeon, result, save);
  for (const g of drops) save.player.gearBag.push(g);
  // 5) 隐藏技能入库：匹配位面 0.1% PRD → rollHiddenSkill → applyHiddenSkill 永久刻印槽位
  if (dungeon.hiddenSkill) applyHiddenSkill(dungeon.hiddenSkill, save);
  // 6) 难度进化：调整 dynFactor（装备强化也会抬升 computePower → 同步进化）
  const { deltaPct } = adjustDynamicFactor(save, result);
  // 7) 首通诸天之心 → 结局演出 + 解锁无尽模式 + 必得 1 个自选隐藏技能
  if (dungeon.plane.id === 'zhutian' && result.victory && !p.firstClear) {
    p.firstClear = true; p.endlessUnlocked = true;
    grantSelfPickHiddenSkill(save);                  // 首通奖励：自选 1 个隐藏技能
  }
  persistSave(save);
}
```

## 十四、编码红线（实现时必须遵守）

1. **位面不设固有强度**：所有敌人数值只由 `基准 × D × 阶段系数 × dynFactor` 生成，禁止按位面手写不同强度表
2. **D 在开副本时快照**：进入副本后不再随玩家局内成长变动（局内 Build 只影响玩家，不影响 D）
3. **不匹配位面零技能**：属性通道的三选一池不得混入任何路线技能，传说技能也只在匹配位面产出。**但装备全位面开放**——属性通道仅对技能封禁，装备掉落必须保留（掉率 ×1.5）
4. **互斥权重 ×0 必须生效**：互斥位面不允许通过随机池出现，兜底逻辑不能只弹窗不重随机
5. **难度系数钳制**：`dynFactor` 永远 ∈ [0.70, 1.50]，调整前必须先 clamp
6. **存档原子写**：所有跨局变更集中在 `finalizeRun` / `activateRoute` 完成后一次 `persistSave`，避免中途退出丢档。装备/技能槽/隐藏技能同属存档字段，禁止写入临时内存
7. **首进固定机关城**：`totalRuns === 0` 判定，全局只有这一个固定分支，此后不得有任何"新手位面保护"
8. **隐藏技能唯一性与永久性**：同一隐藏技能只可获一次（`inventory.hiddenSkills` 去重）；`hidden:true` 槽位不可被局内替换、不可被其他隐藏技能覆盖（除非玩家明确替换，见 applyHiddenSkill）
9. **装备数值闭环**：装备战力加成参与 `computePower` → 抬升 D，禁止把装备加成做成独立"局外碾压"常量，否则破坏难度进化闭环
